/** A real PNG (RGBA, a two-colour diagonal) so that decoding and natural size can be asserted. */
function png(width: number, height: number, a: [number, number, number], b: [number, number, number]) {
  const raw = Buffer.alloc((width * 4 + 1) * height);
  for (let y = 0; y < height; y++) {
    raw[y * (width * 4 + 1)] = 0;
    for (let x = 0; x < width; x++) { const c = x * height > y * width ? a : b, o = y * (width * 4 + 1) + 1 + x * 4; raw[o] = c[0]; raw[o + 1] = c[1]; raw[o + 2] = c[2]; raw[o + 3] = 255; }
  }
  const chunk = (type: string, data: Buffer) => { const body = Buffer.concat([Buffer.from(type), data]), head = Buffer.alloc(4), tail = Buffer.alloc(4); head.writeUInt32BE(data.length); tail.writeUInt32BE(crc32(body)); return Buffer.concat([head, body, tail]); };
  const header = Buffer.alloc(13); header.writeUInt32BE(width, 0); header.writeUInt32BE(height, 4); header[8] = 8; header[9] = 6;
  return Buffer.concat([Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]), chunk("IHDR", header), chunk("IDAT", deflateSync(raw)), chunk("IEND", Buffer.alloc(0))]);
}

/** The stored records of one project exactly as the 1.6.0 App left them in IndexedDB. */
async function stored(page: Page, id: string) {
  return page.evaluate(async designId => {
    const names = (await indexedDB.databases()).map(item => item.name!).filter(Boolean);
    for (const name of names) {
      const db = await new Promise<IDBDatabase>((resolve, reject) => { const open = indexedDB.open(name); open.onsuccess = () => resolve(open.result); open.onerror = () => reject(open.error); });
      try {
        if (!db.objectStoreNames.contains("workbenchDesign")) continue;
        const tx = db.transaction(["workbenchDesign", "workbenchDesignMeta", "workbenchDesignObjects", "workbenchLayouts"]);
        const all = <T,>(store: string) => new Promise<Array<{ key: IDBValidKey; value: T }>>((resolve, reject) => {
          const rows: Array<{ key: IDBValidKey; value: T }> = [], cursor = tx.objectStore(store).openCursor();
          cursor.onsuccess = () => { const at = cursor.result; if (!at) return resolve(rows); rows.push({ key: at.key, value: at.value }); at.continue(); };
          cursor.onerror = () => reject(cursor.error);
        });
        const designs = (await all<{ id: string }>("workbenchDesign")).filter(row => (row.key as string[])[1] === designId);
        if (!designs.length) continue;
        const scope = (designs[0].key as string[])[0];
        const meta = (await all<unknown>("workbenchDesignMeta")).filter(row => (row.key as string[])[1] === designId);
        const objects = (await all<unknown>("workbenchDesignObjects")).filter(row => (row.key as string[])[1] === designId);
        const layout = (await all<{ tabs: Array<{ id: string }>; closed: Array<{ id: string }> }>("workbenchLayouts")).find(row => row.key === scope)?.value;
        const tab = layout && [...layout.tabs, ...layout.closed].find(item => item.id === designId);
        return {
          dbVersion: db.version,
          document: designs[0].value,
          tab,
          meta: Object.fromEntries(meta.map(row => [(row.key as string[])[2], row.value])),
          objects: Object.fromEntries(objects.map(row => [(row.key as string[])[2], row.value])),
        };
      } finally { db.close(); }
    }
    throw new Error("design_not_stored");
  }, id);
}
const text = (file: { base64: string }) => Buffer.from(file.base64, "base64").toString("utf8");
async function save(name: string, value: unknown) { await mkdir(out, { recursive: true }); await writeFile(path.join(out, name), typeof value === "string" || Buffer.isBuffer(value) ? value : JSON.stringify(value)); }

async function renameTab(page: Page, from: string, to: string) {
  await panel(page).getByRole("tab", { name: from, exact: true }).click({ button: "right" });
  await page.getByRole("menu", { name: "标签选项", exact: true }).getByRole("button", { name: "重命名", exact: true }).click();

/** Types into a slide's text in the old editor: the editor itself rewrites and saves the slide file. */
async function retype(page: Page, current: string, next: string) {
  const editor = editorOf(page), target = editor.getByText(current, { exact: true }).first();
  await expect(target).toBeVisible({ timeout: 30000 });
  await target.dblclick(); await page.keyboard.press("Meta+A"); await page.keyboard.type(next); await page.keyboard.press("Escape");
  await expect(editor.getByText(next, { exact: true }).first()).toBeVisible();
}

/** Waits until the old editor has written every pending change into the App's store. */
async function saved(page: Page) {
  await expect(editorOf(page).getByRole("button", { name: /waiting to save/ })).toHaveCount(0, { timeout: 30000 });
  await expect(panel(page).getByRole("status").filter({ hasText: /正在保存|尚未保存/ })).toHaveCount(0, { timeout: 30000 });
}
/** Clicks the 1.6.0 App's own export button and keeps the file it downloads. */
async function backup(page: Page, button: string, name: string) {
  const download = page.waitForEvent("download");
  await panel(page).getByRole("button", { name: button, exact: true }).click();
  const file = await download, bytes = await readFile((await file.path())!);
  expect(file.suggestedFilename()).toMatch(/\.tarobot-design\.json$/);
  expect(JSON.parse(bytes.toString("utf8"))).toMatchObject({ format: "tarobot-design", version: 2 });
  await save(`${name}.tarobot-design.json`, bytes);
  return JSON.parse(bytes.toString("utf8")) as { title: string; document: { files: Record<string, { base64: string; contentType: string }> } };
}

/** A design-system source as earlier versions left it on a device: 1.6.0 lists it under "设计系统" and copies it into a deck. */
async function seedDesignSystem(page: Page, conversationId: string, logo: string, picture: number[]) {
  return page.evaluate(async ({ conversationId, logo, picture }) => {
    const encode = async (bytes: Uint8Array, contentType: string) => {
      let binary = ""; for (const byte of bytes) binary += String.fromCharCode(byte);
      const digest = [...new Uint8Array(await crypto.subtle.digest("SHA-256", bytes))].map(byte => byte.toString(16).padStart(2, "0")).join("");
      return { base64: btoa(binary), contentType, sha256: digest, size: bytes.length };
    };
    const textFile = (value: string, contentType = "text/plain") => encode(new TextEncoder().encode(value), contentType);
    const upload = await encode(new Uint8Array(picture), "image/png"), assetId = upload.sha256.slice(0, 32);
    const tokens = { name: "Brand", color: { themes: [{ id: "light", name: "Light" }], tokens: [
      { name: "coral", value: "#d97757", usage: "Accent" }, { name: "ink", value: "#292723", usage: "Text" }, { name: "paper", value: "#faf9f6", usage: "Background" } ] },
      type: { groups: [{ name: "Display", family: "Georgia", styles: [{ name: "Hero", fontSize: "96px", fontWeight: 600, lineHeight: 1.1 }] }, { name: "Body", family: "Arial", styles: [{ name: "Body", fontSize: "36px" }] }] } };
    const document = { schema: 1, id: "brand", format: "design-system", revision: 3, receipts: [], files: {
      "project/README.md": await textFile(`# Brand\n\nA small house style: coral accent, ink text, paper background.\n\n## Consuming this system\n\n- Tokens: \`project/tokens.json\`\n- Logo: \`project/img/logo.svg\`\n- Sample picture: ![sample](/_blob/${assetId})\n`, "text/markdown"),
      "project/tokens.json": await textFile(JSON.stringify(tokens), "application/json"),
      "project/img/logo.svg": await textFile(logo, "image/svg+xml"),
      "project/Logo.dc.html": await textFile(`<!doctype html><html><head><meta charset="utf-8"></head><body><x-dc><img alt="Sample" src="/_blob/${assetId}"><img alt="Sample again" src="./_blob/${assetId}"></x-dc></body></html>`, "text/html"),
      "project/theme.css": await textFile(`.sample{background-image:url(_blob/${assetId})}`, "text/css"),
      [`project/assets/${assetId}`]: upload,
    } };
    const names = (await indexedDB.databases()).map(item => item.name!).filter(Boolean);
    for (const name of names) {
      const db = await new Promise<IDBDatabase>((resolve, reject) => { const open = indexedDB.open(name); open.onsuccess = () => resolve(open.result); open.onerror = () => reject(open.error); });
      try {
        if (!db.objectStoreNames.contains("workbenchLayouts")) continue;
        const keys = await new Promise<IDBValidKey[]>((resolve, reject) => { const request = db.transaction("workbenchLayouts").objectStore("workbenchLayouts").getAllKeys(); request.onsuccess = () => resolve(request.result); request.onerror = () => reject(request.error); });
        const scope = keys.find(key => typeof key === "string" && JSON.parse(key)[2] === conversationId) as string | undefined;
        if (!scope) continue;
        await new Promise<void>((resolve, reject) => { const tx = db.transaction("workbenchDesign", "readwrite"); tx.objectStore("workbenchDesign").put(document, [scope, "brand"]); tx.oncomplete = () => resolve(); tx.onerror = () => reject(tx.error); });
        return { assetId, document };
      } finally { db.close(); }
    }
    throw new Error("scope_not_found");
  }, { conversationId, logo, picture });
}

    await expect(editor.getByRole("group", { name: "Image, “品牌标志 Brand logo”, 1 of 2", exact: true })).toBeVisible({ timeout: 30000 });
    await expect(editor.getByText(/can’t be edited right now/)).toHaveCount(0);
    await retype(page, "珊瑚色强调、墨色文字、纸色背景。", "珊瑚色强调、墨色文字、纸色背景（手改）。");
    // The old editor reads the installed system's tokens: the paragraph's colour is shown by its token name.
    await editor.getByText("珊瑚色强调、墨色文字、纸色背景（手改）。", { exact: true }).first().click();
    await editor.getByRole("button", { name: "Properties", exact: true }).click();
    await expect(editor.getByRole("button", { name: "Color: ink", exact: true })).toBeVisible();
    await editor.getByRole("tab", { name: "Theme", exact: true }).click();
    await editor.getByRole("button", { name: "Close panel", exact: true }).click();
    await saved(page);
    const exported = await backup(page, "备份 PPT 项目", "design-system-deck");
    // 1.6.0 rewrote the system's /_blob/<id> references to "/project/ds/<ns>/assets/<id>", with a leading slash.
    expect(text(exported.document.files["project/ds/system-brand/Logo.dc.html"])).toContain('src="/project/ds/system-brand/assets/');
    await save("design-system-deck.stored.json", await stored(page, target.tabId));
    expect(errors).toEqual([]);
  } finally { await f.close(); }
});
